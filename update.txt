#!/bin/bash
set -e

echo "=== Update system ==="
sudo apt update

echo "=== Install Yocto dependencies ==="
sudo apt install -y \
    gawk \
    wget \
    git \
    git-lfs \
    diffstat \
    unzip \
    texinfo \
    gcc \
    build-essential \
    chrpath \
    socat \
    cpio \
    python3 \
    python3-pip \
    python3-pexpect \
    python3-git \
    python3-jinja2 \
    python3-subunit \
    xz-utils \
    zstd \
    lz4 \
    file \
    locales \
    iputils-ping \
    xterm \
    rsync \
    bc \
    bison \
    flex \
    libssl-dev \
    libelf-dev \
    libncurses-dev \
    device-tree-compiler \
    repo

echo "=== Configure locale ==="
sudo locale-gen en_US.UTF-8

echo "=== Setup Git ==="
git config --global core.editor vim

echo
echo "Installation finished."
echo
echo "Check:"
echo "  repo --version"
echo "  git --version"
echo "  python3 --version"
``
